---- MODULE ResearchFlow ----
EXTENDS Naturals
VARIABLES phase, gates, approved
vars == <<phase,gates,approved>>
Init == phase="Discovered" /\ gates=FALSE /\ approved=FALSE
Intake == phase="Discovered" /\ phase'="Queued" /\ UNCHANGED <<gates,approved>>
AdHoc == phase="Discovered" /\ phase'= "Scraping" /\ UNCHANGED <<gates,approved>>
Acquire == phase="Queued" /\ phase'="Scraping" /\ UNCHANGED <<gates,approved>>
PacketAccepted == phase="Scraping" /\ phase'="Matching" /\ UNCHANGED <<gates,approved>>
PacketRejected == phase="Scraping" /\ phase'="Needs review" /\ UNCHANGED <<gates,approved>>
Duplicate == phase="Matching" /\ phase'="Duplicate" /\ UNCHANGED <<gates,approved>>
Draft == phase="Matching" /\ phase'="Drafting" /\ UNCHANGED <<gates,approved>>
Validate == phase="Drafting" /\ phase'="Validating" /\ UNCHANGED <<gates,approved>>
Reject == phase="Validating" /\ phase'="Needs review" /\ UNCHANGED <<gates,approved>>
Pass == phase="Validating" /\ phase'="Validated draft" /\ gates'=TRUE /\ UNCHANGED approved
OpenDraft == phase="Validated draft" /\ gates /\ phase'="Draft PR" /\ UNCHANGED <<gates,approved>>
HumanMerge == phase="Draft PR" /\ phase'="Merged content" /\ approved'=TRUE /\ UNCHANGED gates
Dispatch == phase="Merged content" /\ phase'="Site build" /\ UNCHANGED <<gates,approved>>
Deploy == phase="Site build" /\ phase'="Live site (external)" /\ UNCHANGED <<gates,approved>>
BuildFail == phase="Site build" /\ phase'="Build failed" /\ UNCHANGED <<gates,approved>>
Done == phase \in {"Needs review","Duplicate","Live site (external)","Build failed","Validated draft"} /\ UNCHANGED vars
Next == Intake \/ AdHoc \/ Acquire \/ PacketAccepted \/ PacketRejected \/ Duplicate \/ Draft \/ Validate \/ Reject \/ Pass \/ OpenDraft \/ HumanMerge \/ Dispatch \/ Deploy \/ BuildFail \/ Done
Spec == Init /\ [][Next]_vars
PRRequiresGates == phase \in {"Draft PR","Merged content","Site build","Live site (external)"} => gates
LiveRequiresHumanApproval == phase="Live site (external)" => approved
====
