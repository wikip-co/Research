from gmail_reader.app import ensure_db,utc_now_iso,enqueue_publication_job,claim_publication_job,set_publication_job_state

def seed(db,max_attempts=3):

 return enqueue_publication_job(db_path=db,identifier='synthetic',max_attempts=max_attempts)['job']['job_id']
